## 测试规范

### 命名规范

测试文件使用 `test_` 前缀，测试函数使用 `test_` 前缀：

```python
# Good
def test_import_alipay_bill():
    pass

def test_import_fails_with_invalid_file():
    pass

# Bad
def import_test():
    pass

def test1():
    pass
```

### 测试组织

使用类组织相关测试：

```python
class TestBillImport:
    """Bill import related tests."""

    def test_import_alipay(self):
        pass

    def test_import_wechat(self):
        pass

    def test_import_unionpay(self):
        pass
```

### 测试原则

1. **独立性**: 每个测试应该独立运行，不依赖其他测试
2. **可重复性**: 测试结果应该可重复
3. **快速性**: 单元测试应该快速执行
4. **清晰性**: 测试名称应该清楚描述测试内容

## 运行测试

### 基本命令

```bash
# 运行所有测试
python -m pytest

# 运行指定测试文件
python -m pytest tests/test_auth.py

# 运行指定测试函数
python -m pytest tests/test_auth.py::test_login_success

# 显示详细输出
python -m pytest -v

# 显示打印输出
python -m pytest -s
```

### 覆盖率报告

```bash
# 生成覆盖率报告
python -m pytest --cov=. --cov-report=html

# 查看HTML报告
open htmlcov/index.html

# 终端输出覆盖率
python -m pytest --cov=. --cov-report=term
```

### 标记和过滤

```bash
# 运行特定标记的测试
python -m pytest -m "not slow"

# 运行特定表达式
python -m pytest -k "import"
```

import pytest
from fastapi.testclient import TestClient
from web_service.main import app


class TestAuthentication:
    """认证相关测试."""

    def test_register_success(self, client):
        """测试成功注册."""
        response = client.post("/api/auth/register", json={
            "username": "newuser",

        assert response.status_code == 201
        data = response.json()
        assert "access_token" in data

    def test_login_success(self, client, test_user):

### 解析器测试

```python
import pytest
import os
from parsers import get_parser, get_parser_by_platform


class TestParsers:
    """解析器测试."""

    def test_detect_alipay(self, sample_alipay_file):
        """测试支付宝格式检测."""
        parser = get_parser(sample_alipay_file)
        assert parser is not None
        assert parser.get_platform() == "Alipay"

    def test_detect_wechat(self, sample_wechat_file):
        """测试微信格式检测."""
        parser = get_parser(sample_wechat_file)
        assert parser is not None
        assert parser.get_platform() == "WeChatPay"

    def test_parse_alipay(self, sample_alipay_file):
        """测试解析支付宝账单."""
        parser = get_parser(sample_alipay_file)
        df = parser.parse()
        assert len(df) > 0
        assert "amount" in df.columns
        assert "transaction_time" in df.columns

    def test_convert_to_notion_format(self, sample_alipay_file):
        """测试转换为Notion格式."""
        parser = get_parser(sample_alipay_file)
        records = parser.to_notion_format()
        assert len(records) > 0

        record = records[0]
        assert "Name" in record
        assert "Price" in record
        assert "Date" in record
        assert "Income Expense" in record

    def test_filter_non_income_expense(self, sample_alipay_file):
        """测试过滤不计收支记录."""
        parser = get_parser(sample_alipay_file)
        records = parser.to_notion_format()

        # 检查没有"不计收支"的记录
        for record in records:
            income_expense = record.get("Income Expense", {}).get("select", {}).get("name", "")
            assert income_expense != "不计收支"
```

### Mock 测试

```python
from unittest.mock import patch, MagicMock


class TestNotionAPI:
    """Notion API测试."""

    @patch('src.notion_api.NotionClient')
    def test_verify_connection_success(self, mock_client_class):
        """测试连接验证成功."""
        mock_client = MagicMock()
        mock_client.verify_connection.return_value = True
        mock_client_class.return_value = mock_client

        from src.notion_api import NotionClient
        client = NotionClient()
        result = client.verify_connection()
        assert result is True

    @patch('src.notion_api.NotionClient.client.databases.retrieve')
    def test_database_not_found(self, mock_retrieve):
        """测试数据库不存在."""
        from notion_client import APIResponseError

        mock_retrieve.side_effect = APIResponseError(
            "Not found",
            code="object_not_found",
            status=404
        )

        # 测试错误处理
        with pytest.raises(APIResponseError):
            from src.notion_api import NotionClient
            client = NotionClient()
            client.verify_connection()
```
